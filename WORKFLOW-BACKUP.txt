name: Build IPA

on:
  workflow_dispatch:
  push:
    branches: [main, master]

jobs:
  build:
    name: Build unsigned IPA (Release)
    runs-on: macos-15
    timeout-minutes: 45
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Show toolchain
        run: |
          xcodebuild -version
          swift --version
          xcrun --sdk iphoneos --show-sdk-version

      - name: Install XcodeGen
        run: brew install xcodegen

      - name: Generate Xcode project
        run: xcodegen generate

      - name: Build Release for iPhone (without signing)
        run: |
          set -o pipefail
          xcodebuild \
            -project SOBATOVSKY-OS.xcodeproj \
            -scheme SOBATOVSKY-OS \
            -configuration Release \
            -sdk iphoneos \
            -destination 'generic/platform=iOS' \
            -derivedDataPath build \
            CODE_SIGNING_ALLOWED=NO \
            CODE_SIGNING_REQUIRED=NO \
            CODE_SIGN_IDENTITY="" \
            build 2>&1 | tee build.log

      - name: Package IPA and verify its contents
        run: bash scripts/package_ipa.sh

      - name: Upload IPA
        uses: actions/upload-artifact@v4
        with:
          name: SOBATOVSKY-OS-IPA
          path: |
            output/SOBATOVSKY-OS.ipa
            output/IPA-CHECK.txt
          retention-days: 30
          compression-level: 0

      - name: Upload build log
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: build-log
          path: build.log
          retention-days: 14
          if-no-files-found: ignore

  tests:
    name: Unit tests (simulator, non-blocking)
    runs-on: macos-15
    continue-on-error: true
    timeout-minutes: 45
    steps:
      - uses: actions/checkout@v4
      - name: Install XcodeGen
        run: brew install xcodegen
      - name: Generate Xcode project
        run: xcodegen generate
      - name: Run tests
        run: |
          set -o pipefail
          DEVICE=$(xcrun simctl list devices available | grep -m1 -oE 'iPhone [^(]+' | sed 's/[[:space:]]*$//')
          echo "Simulator: $DEVICE"
          xcodebuild \
            -project SOBATOVSKY-OS.xcodeproj \
            -scheme SOBATOVSKY-OS \
            -destination "platform=iOS Simulator,name=$DEVICE" \
            CODE_SIGNING_ALLOWED=NO \
            test 2>&1 | tee test.log
      - name: Upload test log
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: test-log
          path: test.log
          retention-days: 14
          if-no-files-found: ignore
